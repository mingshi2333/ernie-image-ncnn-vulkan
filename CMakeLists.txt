cmake_minimum_required(VERSION 3.19)
project(ernie_image_ncnn_vulkan VERSION 0.1.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(ERNIE_ENABLE_VULKAN "Build Vulkan support" ON)
option(ERNIE_MINIMAL_NCNN "Build only layers needed by current probes" ON)
option(ERNIE_ALLOW_UNPINNED_NCNN "Allow an explicitly selected experimental ncnn revision" OFF)
option(ERNIE_BUILD_TOKENIZER "Build the native tokenizer bridge (requires Cargo)" OFF)
set(ERNIE_NCNN_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/third_party/ncnn" CACHE PATH "ncnn source checkout")

file(READ "${CMAKE_CURRENT_SOURCE_DIR}/sources.lock.json" sources_lock)
string(JSON ncnn_revision GET "${sources_lock}" ncnn revision)
if(NOT EXISTS "${ERNIE_NCNN_SOURCE_DIR}/src/net.h")
    message(FATAL_ERROR "Initialize ncnn with: git submodule update --init third_party/ncnn")
endif()

if(ERNIE_BUILD_TOKENIZER)
    find_program(CARGO_EXECUTABLE cargo REQUIRED)
    find_package(Threads REQUIRED)
    set(tokenizer_library "${CMAKE_BINARY_DIR}/cargo/release/libernie_tokenizer_bridge.a")
    add_custom_command(OUTPUT "${tokenizer_library}"
        COMMAND "${CMAKE_COMMAND}" -E env CARGO_BUILD_JOBS=4 "${CARGO_EXECUTABLE}" build --release --locked
                --manifest-path "${CMAKE_CURRENT_SOURCE_DIR}/tokenizer/Cargo.toml" --target-dir "${CMAKE_BINARY_DIR}/cargo"
        DEPENDS tokenizer/Cargo.toml tokenizer/Cargo.lock tokenizer/src/lib.rs
        COMMENT "Building pinned native Tokenizers bridge")
    add_custom_target(ernie-tokenizer-rust DEPENDS "${tokenizer_library}")
    add_library(ernie-tokenizer-native STATIC IMPORTED)
    set_target_properties(ernie-tokenizer-native PROPERTIES IMPORTED_LOCATION "${tokenizer_library}")
    add_dependencies(ernie-tokenizer-native ernie-tokenizer-rust)
    add_library(ernie-tokenizer STATIC src/tokenizer.cpp)
    target_include_directories(ernie-tokenizer PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src")
    target_link_libraries(ernie-tokenizer PUBLIC ernie-tokenizer-native Threads::Threads ${CMAKE_DL_LIBS} m)
    add_executable(ernie-tokenize probes/tokenizer_runner.cpp)
    target_link_libraries(ernie-tokenize PRIVATE ernie-tokenizer)
endif()
find_package(Git REQUIRED)
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${ERNIE_NCNN_SOURCE_DIR}" rev-parse HEAD
    OUTPUT_VARIABLE actual_revision OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT actual_revision STREQUAL ncnn_revision AND NOT ERNIE_ALLOW_UNPINNED_NCNN)
    message(FATAL_ERROR "ncnn revision differs from sources.lock.json: ${actual_revision}")
endif()

set(NCNN_VULKAN ${ERNIE_ENABLE_VULKAN} CACHE BOOL "" FORCE)
set(NCNN_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(NCNN_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(NCNN_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(NCNN_BUILD_BENCHMARK OFF CACHE BOOL "" FORCE)
set(NCNN_INSTALL_SDK OFF CACHE BOOL "" FORCE)
if(ERNIE_MINIMAL_NCNN)
    # RMSNorm Vulkan shares reduction shaders with LayerNorm.
    # Extend this list when new exported components require additional layers.
    file(STRINGS "${ERNIE_NCNN_SOURCE_DIR}/src/CMakeLists.txt" layer_lines REGEX "^ncnn_add_layer\\(")
    set(probe_layers input split cast packing flatten softmax gemm sdpa memorydata unaryop reduction binaryop reshape permute rmsnorm layernorm slice expanddims concat squeeze gelu convolution padding innerproduct swish crop)
    foreach(layer_line IN LISTS layer_lines)
        string(REGEX REPLACE "^ncnn_add_layer\\(([A-Za-z0-9_]+).*" "\\1" layer_name "${layer_line}")
        string(TOLOWER "${layer_name}" layer_name)
        if(layer_name IN_LIST probe_layers)
            set(WITH_LAYER_${layer_name} ON CACHE BOOL "" FORCE)
        else()
            set(WITH_LAYER_${layer_name} OFF CACHE BOOL "" FORCE)
        endif()
    endforeach()
endif()
add_subdirectory("${ERNIE_NCNN_SOURCE_DIR}" "${CMAKE_BINARY_DIR}/ncnn")

add_executable(ernie-attention-probe probes/attention_probe.cpp)
target_link_libraries(ernie-attention-probe PRIVATE ncnn)
target_compile_definitions(ernie-attention-probe PRIVATE ERNIE_NCNN_REVISION="${actual_revision}")
enable_testing()
add_test(NAME native_cache_cpu COMMAND ernie-attention-probe --backend cpu)
if(ERNIE_ENABLE_VULKAN)
    add_test(NAME native_cache_vulkan COMMAND ernie-attention-probe --backend vulkan)
    set_tests_properties(native_cache_vulkan PROPERTIES SKIP_RETURN_CODE 77)
endif()

add_library(ernie-runtime STATIC src/ernie_gelu.cpp src/ernie_rmsnorm.cpp src/latent_ops.cpp src/block_sequence.cpp src/dit.cpp)
target_link_libraries(ernie-runtime PUBLIC ncnn)
target_include_directories(ernie-runtime PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src")
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    # Euler and BN follow the reference's separate multiply/add operations.
    set_source_files_properties(src/latent_ops.cpp PROPERTIES COMPILE_OPTIONS "-ffp-contract=off")
endif()
add_executable(ernie-block-runner probes/block_runner.cpp)
target_link_libraries(ernie-block-runner PRIVATE ernie-runtime)
if(ERNIE_ENABLE_VULKAN)
    target_sources(ernie-block-runner PRIVATE probes/attention_diagnostic.cpp)
    target_include_directories(ernie-block-runner PRIVATE "${ERNIE_NCNN_SOURCE_DIR}/src/layer")
endif()
add_executable(ernie-gelu-probe probes/gelu_probe.cpp)
target_link_libraries(ernie-gelu-probe PRIVATE ernie-runtime)
add_executable(ernie-latent-runner probes/latent_runner.cpp)
target_link_libraries(ernie-latent-runner PRIVATE ernie-runtime)
add_executable(ernie-block-sequence-runner probes/block_sequence_runner.cpp)
target_link_libraries(ernie-block-sequence-runner PRIVATE ernie-runtime)
add_executable(ernie-head-runner probes/head_runner.cpp)
target_link_libraries(ernie-head-runner PRIVATE ernie-runtime)
add_executable(ernie-latent-contract tests/test_latent_contract.cpp)
target_link_libraries(ernie-latent-contract PRIVATE ernie-runtime)
add_test(NAME latent_contract_cpu COMMAND ernie-latent-contract)
add_executable(ernie-rmsnorm-probe probes/rmsnorm_probe.cpp)
target_link_libraries(ernie-rmsnorm-probe PRIVATE ernie-runtime)
add_test(NAME rmsnorm_range_cpu COMMAND ernie-rmsnorm-probe cpu fp32)
add_test(NAME layernorm_range_cpu COMMAND ernie-rmsnorm-probe cpu fp32 --layernorm)
add_test(NAME erf_gelu_cpu COMMAND ernie-gelu-probe cpu fp32)
if(ERNIE_ENABLE_VULKAN)
    foreach(precision fp32 fp16 bf16)
        add_test(NAME erf_gelu_vulkan_${precision} COMMAND ernie-gelu-probe vulkan ${precision})
        set_tests_properties(erf_gelu_vulkan_${precision} PROPERTIES SKIP_RETURN_CODE 77)
        add_test(NAME rmsnorm_range_vulkan_${precision} COMMAND ernie-rmsnorm-probe vulkan ${precision})
        set_tests_properties(rmsnorm_range_vulkan_${precision} PROPERTIES SKIP_RETURN_CODE 77)
        add_test(NAME layernorm_range_vulkan_${precision} COMMAND ernie-rmsnorm-probe vulkan ${precision} --layernorm)
        set_tests_properties(layernorm_range_vulkan_${precision} PROPERTIES SKIP_RETURN_CODE 77)
    endforeach()
endif()
