cmake_minimum_required(VERSION 3.19)
project(ernie_image_ncnn_vulkan VERSION 0.1.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
# Keep the documented command paths stable when targets live in subdirectories.
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
include(CTest)
option(ERNIE_ENABLE_VULKAN "Build Vulkan support" ON)
option(ERNIE_EXPERIMENT_MAPPED_MODEL_LOADING "Request ncnn mapped model loading in the generation pipeline" OFF)
option(ERNIE_MINIMAL_NCNN "Build only layers required by ERNIE" ON)
option(ERNIE_ALLOW_UNPINNED_NCNN "Allow an explicitly selected experimental ncnn revision" OFF)
option(ERNIE_BUILD_TOKENIZER "Build the native tokenizer bridge (requires Cargo)" OFF)
option(ERNIE_BUILD_GENERATOR "Build the native pipeline library and PNG CLI" OFF)
option(ERNIE_INSTALL_SDK "Install the public C++ pipeline and its static dependencies" OFF)
option(ERNIE_BUILD_PROBES "Build model conversion and diagnostic runners" ${BUILD_TESTING})
set(ERNIE_NCNN_SOURCE_DIR "${PROJECT_SOURCE_DIR}/third_party/ncnn" CACHE PATH "ncnn source checkout")
include(GNUInstallDirs)
if(ERNIE_INSTALL_SDK AND NOT ERNIE_BUILD_GENERATOR)
    message(FATAL_ERROR "ERNIE_INSTALL_SDK requires ERNIE_BUILD_GENERATOR=ON")
endif()

include(cmake/Dependencies.cmake)
if(ERNIE_BUILD_TOKENIZER)
    add_subdirectory(tokenizer)
endif()
add_subdirectory(src)
if(ERNIE_BUILD_GENERATOR)
    add_subdirectory(cli)
endif()
if(ERNIE_BUILD_PROBES OR BUILD_TESTING)
    add_subdirectory(probes)
endif()
if(BUILD_TESTING)
    add_subdirectory(tests)
endif()
if(ERNIE_INSTALL_SDK)
    include(cmake/Packaging.cmake)
endif()
